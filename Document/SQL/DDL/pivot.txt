with base_query as (
select
	tr.tbl_row_sequene seq,
	tr.tbl_col_name column,
	tr.tbl_row_value_number number,
	tr.tbl_row_value_string string
from
	tbl_dtls.table_name tn,
	tbl_dtls.table_column tc,
	tbl_dtls.table_row tr
where
	tn.user_id = '12345'
	and tn.tbl_name = 't_product'
	and tc.tbl_id = tn.tbl_id
	and tr.tbl_id = tn.tbl_id
	and tr.tbl_col_id = tc.tbl_col_id
	and tr.effective_date is null
	and tc.effective_date is null
order by
	tr.tbl_row_sequene 
)
select
	qry.seq "seq",
	MIN( case when qry.column = 'product_id' then qry.number end ) "product_id",
	MIN( case when qry.column = 'product_name' then qry.string end ) "product_name",
	MIN( case when qry.column = 'price' then qry.number end ) "price",
	MIN( case when qry.column = 'stocks_count' then qry.number end ) "stocks_count", 
	MIN( case when qry.column = 'description' then qry.string end ) "description"
from
	base_query qry
group by
	qry.seq
